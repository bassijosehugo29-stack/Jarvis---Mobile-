cmake_minimum_required(VERSION 3.22.1)
project(jarvis_conversation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
    # Baseline emulator CPU; never emit host AVX instructions into an Android library.
    set(GGML_AVX OFF CACHE BOOL "" FORCE)
    set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
    set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
    set(GGML_FMA OFF CACHE BOOL "" FORCE)
    set(GGML_F16C OFF CACHE BOOL "" FORCE)
    set(GGML_SSE42 OFF CACHE BOOL "" FORCE)
endif()
include(FetchContent)
# Immutable upstream source; no package manager, remote inference or runtime code download.
FetchContent_Declare(llama
    URL https://github.com/ggml-org/llama.cpp/archive/74d4f5b041ad837153b0e90fc864b8290e01d8d5.tar.gz
)
FetchContent_MakeAvailable(llama)
add_library(jarvis_conversation SHARED jarvis_conversation.cpp)
target_link_libraries(jarvis_conversation PRIVATE llama android log)
target_link_options(jarvis_conversation PRIVATE "-Wl,-z,max-page-size=16384")
